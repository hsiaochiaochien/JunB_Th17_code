#==========================================================================================================================================================
step1: The read length below 30, quality below 25 and Nextera adaptor sequences were removed  

module load cutadapt/2.8 

ilist_R1=(/work/IshikawaU/Hsiaochiao/ATACseq20200123/rawdata/*R1*.fastq.gz) 
ilist_R2=(/work/IshikawaU/Hsiaochiao/ATACseq20200123/rawdata/*R2*.fastq.gz)

input_R1=${ilist_R1[$SLURM_ARRAY_TASK_ID]}
input_R2=${ilist_R2[$SLURM_ARRAY_TASK_ID]}

output_forward=$(basename -s _001.fastq.gz $input_R1).fq.gz
output_reverse=$(basename -s _001.fastq.gz $input_R2).fq.gz

ADAPTER_FWD=(CTGTCTCTTATACACATCTCCGAGCCCACGAGAC)
ADAPTER_REV=(CTGTCTCTTATACACATCTGACGCTGCCGACGA)

cutadapt -q 25 -a $ADAPTER_FWD -A $ADAPTER_REV --minimum-length 30 -o $output_forward -p $output_reverse $input_R1 $input_R2


#==========================================================================================================================================================
step2: These reads were aligned to a reference genome mm10 

module load bowtie2/2.3.4.3

ilist_R1=(/work/IshikawaU/Hsiaochiao/ATACseq20200123/process/2.1_cutadapt/*_R1.fq.gz)
ilist_R2=(/work/IshikawaU/Hsiaochiao/ATACseq20200123/process/2.1_cutadapt/*_R2.fq.gz)

input_R1=${ilist_R1[$SLURM_ARRAY_TASK_ID]}
input_R2=${ilist_R2[$SLURM_ARRAY_TASK_ID]}

output=$(basename -s _R1.fq.gz $input_R1)

bowtie2 \
-X 2000 \
--very-sensitive \
--threads 4 \
-x /work/IshikawaU/Hsiaochiao/mm10/Mus_musculus/UCSC/mm10/Sequence/Bowtie2Index/genome \
-1 $input_R1 \
-2 $input_R2 \
-S $output.sam


#==========================================================================================================================================================
step3 : Removal of mitochondrial reads. 

module load samtools/1.9

ilist=(/flash/IshikawaU/HsiaoChiao/ATACseq20200123/process/03_Bowtie2_cutadapt/*.sam)
input=${ilist[$SLURM_ARRAY_TASK_ID]}
output=$(basename -s .sam $input)_mapped.bam

grep -v 'chrM' $input | samtools view -b -h -q 30 -F 4 -f 0x2 -> $output

#==========================================================================================================================================================
step4: sort sam and conver to bam

module load picard/2.7.0

ilist=(/flash/IshikawaU/HsiaoChiao/ATACseq20200123/process/04_bam_Bowtie2_cutadapt/*.bam)
input_bam=${ilist[$SLURM_ARRAY_TASK_ID]}
sorted_bam=$(basename -s _mapped.bam $input_bam)_sorted.bam

picard SortSam \
I=$input_bam \
O=$sorted_bam \
SORT_ORDER=coordinate


#==========================================================================================================================================================
#step5: Duplicate reads were removed using the Picard
#the bam files can be download

module load picard/2.7.0

ilist=(/flash/IshikawaU/HsiaoChiao/ATACseq20200123/process/05_sort_bam_Bowtie2_cutadapt/*.bam)
